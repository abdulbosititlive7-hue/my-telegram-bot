def is_system_admin(user_id: int) -> bool:
    cursor.execute(
        "SELECT user_id FROM system_admins WHERE user_id = ?", (user_id,)
    )
    return cursor.fetchone() is not None


def get_admin_keyboard():

@router.message(F.text == "?? Foydalanuvchilar")
async def show_users_list(message: Message):
    if not is_system_admin(message.from_user.id):
        return

    cursor.execute("SELECT user_id, full_name, username FROM applications")
    users = cursor.fetchall()

    if not users:
        await message.answer("Botda hali hech kim ro'yxatdan o'tmagan.")
        return

    text = f"?? <b>Botdagi barcha foydalanuvchilar ({len(users)} ta):</b>\n\n"
    for idx, u in enumerate(users, start=1):
        username_str = f"@{u[2]}" if u[2] else "Username yo'q"
        text += f"{idx}. <b>{u[1]}</b> | {username_str}\n?? ID: <code>{u[0]}</code>\n---\n"

    await message.answer(text, parse_mode="HTML")


# ADMIN REPLY: /send <matn> orqali xabar yuborish
@router.message(F.reply_to_message, F.text.startswith("/send"))
async def send_custom_message(message: Message):
    if not is_system_admin(message.from_user.id):
        return

    parts = message.text.split(maxsplit=1)
    if len(parts) < 2:
        await message.answer("?? Iltimos, yubormoqchi bo'lgan matningizni kiriting.\nMasalan: <code>/send Salom!</code>", parse_mode="HTML")
        return

    text_to_send = parts[1]
    reply_text = message.reply_to_message.text or ""

    target_id = None
    if "ID:" in reply_text or "ID: " in reply_text:
        try:
            target_id = int(reply_text.split("ID:")[1].split()[0].replace("</code>", "").strip())
        except Exception:
            target_id = None

    if target_id:
        try:
            await bot.send_message(
                target_id,
                f"?? <b>Admindan xabar:</b>\n\n{text_to_send}",
                parse_mode="HTML"
            )
            await message.answer("? Xabar foydalanuvchiga muvaffaqiyatli yuborildi!")
        except Exception as e:
            await message.answer(f"? Xabar yuborishda xatolik yuz berdi: {e}")
    else:
        await message.answer("?? ID topilmadi. Javob bering (reply qiling) va xabarda ID raqami borligiga ishonch hosil qiling.")
